\section{Executive Summary}
%==============================================================================

\subsection{What Zinger Is}
Zinger is a TypeScript full-stack system (\texttt{/home/david/Zinger}) comprising:
\begin{itemize}[noitemsep]
  \item A \textbf{scan loop} on Polymarket CLOB/Gamma for BTC and ETH 5m binaries
  \item Two engines: \textbf{directional} (TA + book + optional strike forecast) and \textbf{atomic arb} (buy both legs when UP+DOWN $<$ \$1 minus fees)
  \item \textbf{Paper and live} modes with operator dashboard (port 3000)
  \item \textbf{Governor} regime switching, edge gate, signal health, and data assurance layers
\end{itemize}

\subsection{Arb Strategy (How It Works)}
\begin{enumerate}[noitemsep]
  \item When $\text{ask}_{\text{UP}} + \text{ask}_{\text{DOWN}} < 1 - \text{fees} - \text{margin}$, buy equal shares of both legs
  \item Walk both ask ladders --- touch gap often overstates real fill
  \item Hold both to window settlement; one leg pays \$1/share, other \$0
  \item Package net $=$ locked edge at entry (typically \$0.25--\$0.65 per package on recent fills)
  \item If one leg fails to fill, rollback the other (costs both fees)
\end{enumerate}

\section{Known Problems \& Failure Modes}
%==============================================================================

\begin{enumerate}
  \item \textbf{Directional edge not demonstrated on paper.} Repeated SL hits in 30¢ band; fees $\sim$3--7\% of notional erode small targets.
  \item \textbf{Gate stacking (historical).} DD breaker + edge gate + signal health + orphan blocks created long no-trade periods. Partially fixed; still complex.
  \item \textbf{Live arb rarity.} Real CLOB gaps after ladder walk are infrequent; 2 arb packages in hours vs 50+ directional trades.
  \item \textbf{Synthetic backtest optimism gap.} 6mo +0.6\% on injected mispricing may not transfer to live microstructure.
  \item \textbf{Spot vs Chainlink resolution feed.} Strike forecast can disagree with market when spot feed diverges from settlement oracle.
  \item \textbf{UP-side bias (active).} Session side balance 93\% UP (42 vs 3). Side-balance rule requires both outcomes passing edge filter; if DOWN rarely passes, monopoly persists.
  \item \textbf{Process stability.} OOM kills (exit 137), scan hangs, orphan legs --- mitigated but recurred.
  \item \textbf{Hold-to-settle dead band (historical).} 0.42--0.50 neither underdog nor favorite caused unintended mid-window SL exposure.
\end{enumerate}

\section{What to Improve (Prioritized Roadmap)}
%==============================================================================

\subsection{P0 --- Prove or Kill Directional Edge}
\begin{itemize}
  \item Run isolated \textbf{directional-only} and \textbf{arb-only} sessions (100+ closes each) with clean resets
  \item Measure: expectancy, fee drag, SL rate, time-in-trade, side balance
  \item If 30¢ band WR $<$ break-even after fees $\rightarrow$ widen band, tighten confidence gate, or pause directional
  \item Decision rule: no live capital until paper expectancy $>$ 0 over 200 closes in chosen profile
\end{itemize}

\subsection{P1 --- Real Book Data for Arb}
\begin{itemize}
  \item Record CLOB WS snapshots to disk (\texttt{data/clob-snapshots/})
  \item Replay recorded books in backtest instead of synthetic \texttt{bookSim} gaps
  \item Calibrate \texttt{minArbGap} / \texttt{arbMinMarginPct} on real gap distribution
  \item Target: know live arb hit rate (packages/day) before sizing up
\end{itemize}

\subsection{P1 --- Fee-Adjusted Expectancy Dashboard}
\begin{itemize}
  \item Package-level arb P\&L card (not per-leg) in UI
  \item Per-engine rolling expectancy, break-even WR, Kelly validity
  \item Alert when directional WR below break-even for 20+ closes
\end{itemize}

\subsection{P2 --- Resolution Feed Alignment}
\begin{itemize}
  \item Source spot from same feed Polymarket uses for PTB resolution (Chainlink stream)
  \item Reduce false strike-forecast vetoes and wrong-side entries on feed skew
\end{itemize}

\subsection{P2 --- Simplify Gate Stack}
\begin{itemize}
  \item Single ``can trade directional / can trade arb'' panel with explicit reasons
  \item Preset profiles: \texttt{arb-hunt}, \texttt{directional-30c}, \texttt{classic-paper} --- one-click, no conflicting overlays
  \item Edge gate: disable or raise \texttt{edgeMinTrades} on paper (currently misleading ``arb-only'' alerts)
\end{itemize}

\subsection{P3 --- Reliability}
\begin{itemize}
  \item Memory profiling / OOM guardrails on scan loop
  \item Automated orphan detection + unwind cron
  \item Health endpoint: last scan age, \texttt{\_scanning} lock state
\end{itemize}

\subsection{P3 --- Live Readiness}
\begin{itemize}
  \item CTF instant merge path tested on testnet
  \item Live safety caps enforced (\texttt{assertLiveSafetyCaps})
  \item Arb-only live pilot with \$50 cap before directional live
\end{itemize}
