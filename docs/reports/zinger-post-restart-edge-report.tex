\subsection{Why we restarted}
\begin{enumerate}[leftmargin=*,itemsep=0.35em]
  \item \textbf{Stale scan / stuck scan lock} --- last scan age exceeded watchdog threshold ($\sim$100s); buys blocked via \texttt{scan\_fresh}.
  \item \textbf{False orphan on 4h} --- \texttt{dataAssurance} treated every slug as a 300s (5m) window. A live \texttt{btc-updown-4h-*} directional fill was flagged ``past window end'' minutes after open and blocked \emph{all} buys via \texttt{orphan\_paper}.
\end{enumerate}

\subsection{Fixes shipped at restart}
\begin{itemize}[leftmargin=*,itemsep=0.25em]
  \item Parse window length from slug (\texttt{5m}=300s, \texttt{15m}=900s, \texttt{4h}=14400s) in \texttt{dataAssurance.ts}.
  \item Process restart on \texttt{:3000}; hybrid re-arm via \texttt{scripts/arm-hybrid-fills.mjs}.
  \item Keep slice knobs: \texttt{arbSliceUsd=\$15}, \texttt{maxArbPerSlug=6}, \texttt{maxArbPackages=16}.
\end{itemize}

  \item \textbf{Arb is the steady engine} in the pre-restart wait window --- multi-asset (ETH/SOL/XRP) with \$15-style bites and merge exits.

\end{enumerate}
