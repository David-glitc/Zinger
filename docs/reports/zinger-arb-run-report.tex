\subsection{How Arb Works (Reminder)}
\begin{enumerate}[noitemsep]
  \item When $\text{ask}_{\text{UP}} + \text{ask}_{\text{DOWN}} < 1 - \text{fees} - \text{margin}$, buy equal shares of both legs
  \item Walk both ask ladders (not touch-only)
  \item Hold to settlement; package net P\&L is locked at entry
  \item Individual legs show large win/loss at settle --- \textbf{package net} is what matters
\end{enumerate}

%==============================================================================
\section{What to Watch (Live Arb Session)}
%==============================================================================

\begin{itemize}
  \item \textbf{Packages per day} --- live books are thinner than synthetic; expect far fewer than 44/61d backtest rate unless gaps appear
  \item \textbf{Package-level P\&L} --- not per-leg settlement display
  \item \textbf{Orphan legs} --- if one leg fills and other fails, rollback should unwind (costs both fees)
  \item \textbf{Scan freshness} --- stale scan alert means hung loop; restart bot if $>$90s
  \item \textbf{DD breaker} --- at 15\% off peak, governor may force arb-only anyway (already arb-only)
\end{itemize}
